test('the canonical local-access seed transaction rolls back earlier principals and grants after a later invalid seed',async t=>{
  const root=await mkdtemp(join(tmpdir(),'q4-seed-'));
  t.after(()=>rm(root,{recursive:true,force:true,maxRetries:2,retryDelay:50}));
  const accessPath=join(root,'seed-rollback.sqlite'),instanceId=`runtime-${randomUUID()}`;
  const access=new LocalAccess(accessPath,{productId:'service-desk',instanceId});
  try{
    assert.throws(()=>access.seed([
      {id:'admin',password:'Valid-Admin-Password-001',grants:[{tenantId:'customer-a',permissions:['desk.read']}]},
      {id:'invalid-viewer',password:'short',grants:[{tenantId:'customer-a',permissions:['desk.read']}]}
    ]),/at least 12 characters/);
    assert.equal(access.store.db.prepare('SELECT COUNT(*) AS count FROM principals').get().count,0,'a later seed failure rolls back the first inserted principal');
    assert.equal(access.store.db.prepare('SELECT COUNT(*) AS count FROM grants').get().count,0,'the corresponding grants roll back in the same transaction');
    assert.equal(access.store.db.prepare('SELECT COUNT(*) AS count FROM sessions').get().count,0);
    assert.deepEqual({...access.store.db.prepare('SELECT product,instance FROM instance_binding').get()},{product:'service-desk',instance:instanceId},
      'the canonical constructor binding remains independent of the failed seed transaction');
  }finally{access.close();}
});

test('an owned process close releases SQLite after the canonical local-access constructor faults',async t=>{
  const temp=await mkdtemp(join(tmpdir(),'q4-access-close-')),databasePath=join(temp,'access.sqlite');
  let child,childClosed=false,closePromise;
  t.after(async()=>{
    if(child&&!childClosed){try{child.kill();await waitBounded(closePromise,'constructor fixture process close',5000);}catch{}}
    if(child&&!childClosed)throw new Error(`Constructor-failure fixture retained at ${temp}; its owner process close was not observed`);
    await rm(temp,{recursive:true,force:true,maxRetries:2,retryDelay:50});
  });
  const instanceId=`runtime-${randomUUID()}`;
  const initial=new LocalAccess(databasePath,{productId:'service-desk',instanceId});

  const setup=new DatabaseSync(databasePath);

  finally{setup.close();}

  }finally{reopened.close();}

async function waitForBuild(store,id,timeoutMs=90000) {
  const end=Date.now()+timeoutMs;
  while(Date.now()<end) {
    const row=store.build(id);
    if(row&&['SUCCEEDED','FAILED','CANCELLED'].includes(row.state))return row;
    await new Promise(resolve=>setTimeout(resolve,50));
  }
  throw new Error(`Build ${id} did not settle before the bounded fixture deadline`);
}

  const externalOwners=new Set();
  t.after(async()=>{
    const cleanupErrors=[];
    for(const owner of externalOwners) {
      if(!owner.closed) {
        try { owner.go?.();owner.release?.();await owner.waitForClose?.(); }
        catch(error) { cleanupErrors.push(error); }
      }
    }
    const unprovenOwners=[...externalOwners].filter(owner=>!owner.closed||!owner.safeCleanup);
    let runtimeSettled=true;
    try { await runtimes?.close(); }
    catch(error) {
      const expected=expectedUnknownQuarantine;

      const slot=expected?store?.managedRuntimeSlot():undefined;
      const slotOwner=expected?store?.db.prepare('SELECT instance_id AS instanceId,operation_id AS operationId FROM runtime_physical_slot WHERE singleton=1').get():undefined;

        attempt.helperClosed===true&&attempt.executionClean===false&&slot?.state==='UNKNOWN'&&slot.instanceId===expected.instanceId&&
        slotOwner?.instanceId===expected.instanceId&&slotOwner?.operationId===expected.operationId;

async function pairedFetch(server,path,{method='GET',body,cookie}={}) {
  const headers={Origin:server.url};
  if(cookie)headers.Cookie=cookie;
  if(body!==undefined)headers['Content-Type']='application/json';
  return fetch(server.url+path,{method,headers,body:body===undefined?undefined:JSON.stringify(body)});
}
async function waitBounded(signal,label,timeoutMs=10000) {
  let timer;
  try {
    return await Promise.race([signal,new Promise((_,reject)=>{timer=setTimeout(()=>reject(new Error(`${label} did not arrive before the fixture deadline`)),timeoutMs);})]);
  } finally { clearTimeout(timer); }
}

async function runOneShotRuntimeScript(context,scriptPath,args,expectedExitCode) {
  const profile=assertWindowsBuildHost(ROOT,ROOT);
  assert.ok(profile,'the selected pinned Windows PowerShell profile is available');
  const helperPath=join(ROOT,'tools','build-job.ps1');
  const helperDigest=createHash('sha256').update(await readFile(helperPath)).digest('hex');
  assert.equal(helperDigest,'24364492a899af03afde4ef023d47e67744c2db0fa979208eaa0137214e68ea9','the fixed native helper matches its reviewed pin');
  const operationRoot=join(context.temp,`native-${randomUUID()}`);
  await mkdir(operationRoot);
  const nonce=randomBytes(24).toString('base64url');
  const logPath=join(operationRoot,'private.log');
  const child=spawn(profile.path,['-NoLogo','-NoProfile','-NonInteractive','-File',helperPath,'-RunnerNonce',nonce,
    '-NodeExe',process.execPath,'-WorkingDirectory',operationRoot,'-ScriptPath',scriptPath,'-ArgumentsJson',JSON.stringify(args),
    '-LogPath',logPath,'-RuntimeProtocol'],{cwd:operationRoot,env:Object.fromEntries(['PATH','SystemRoot','WINDIR','TEMP','TMP','COMSPEC','PATHEXT']
      .filter(key=>process.env[key]).map(key=>[key,process.env[key]])),shell:false,windowsHide:true,stdio:['pipe','pipe','pipe']});
  const owner={child,closed:false,safeCleanup:false,stdout:'',stderr:'',lines:[],ready:false,nonce,operationRoot};
  owner.closedPromise=new Promise(resolve=>child.once('close',(code,signal)=>{owner.closed=true;owner.closeValue={code,signal};resolve(owner.closeValue);}));
  child.once('error',error=>{owner.spawnError=error;});
  child.stdout.on('data',chunk=>{
    owner.stdoutBytes=(owner.stdoutBytes??0)+chunk.length;
    if(owner.stdoutBytes>4096){owner.protocolError=true;return;}
    owner.stdout=(owner.stdout+chunk.toString('ascii')).slice(-4096);
    for(;;){const at=owner.stdout.indexOf('\n');if(at<0)break;let line=owner.stdout.slice(0,at);owner.stdout=owner.stdout.slice(at+1);if(line.endsWith('\r'))line=line.slice(0,-1);owner.lines.push(line);}
  });
  child.stderr.on('data',chunk=>{owner.stderrBytes=(owner.stderrBytes??0)+chunk.length;});
  owner.resumeSent=false;
  owner.go=()=>{if(!owner.ready||owner.closed||owner.resumeSent)return;owner.resumeSent=true;try{child.stdin.write(`RESUME:${nonce}\n`,'ascii');}catch(error){owner.controlError=error;}};
  owner.release=()=>{};
  child.stdin.on('error',error=>{owner.controlError=error;});
  owner.waitForClose=async()=>{
    const close=await waitBounded(owner.closedPromise,'fixed native helper close',20000);
    const match=owner.lines.length===2&&owner.lines[0]===`QB_READY:${nonce}`&&new RegExp(`^QB_EXIT:${nonce}:(\\d+):EMPTY$`).exec(owner.lines[1]);
    if(!owner.protocolError&&match&&Number(match[1])===close.code&&close.signal===null)owner.safeCleanup=true;
    return close;
  };
  owner.waitForReady=async()=>{
    const deadline=Date.now()+15000;
    while(Date.now()<deadline){
      if(owner.lines.includes(`QB_READY:${nonce}`)){owner.ready=true;return;}
      if(owner.closed)break;
      await new Promise(resolve=>setTimeout(resolve,20));
    }
    throw new Error('The fixed native helper did not report its nonce-bound assignment before the fixture deadline');
  };
  owner.prove=async()=>{
    const close=await owner.waitForClose();
    assert.equal(owner.protocolError,undefined,'native helper stdout remained inside the protocol bound');
    assert.equal(owner.lines.length,2,'native helper emitted exactly assignment and exit proof');
    assert.equal(owner.lines[0],`QB_READY:${nonce}`);
    const match=new RegExp(`^QB_EXIT:${nonce}:(\\d+):EMPTY$`).exec(owner.lines[1]);
    assert.ok(match,'native helper exit proof uses the exact nonce and EMPTY marker');
    const privateOutput=await readFile(logPath,'utf8').catch(()=>''),stage=/^NATIVE_FAILURE:([a-z-]+)$/m.exec(privateOutput)?.[1];
    const boundedOutcome=stage?`NATIVE_FAILURE:${stage}`:'no fixed fixture failure marker';
    assert.equal(Number(match[1]),expectedExitCode,`the fixed child returned an unexpected exit code (${boundedOutcome})`);
    assert.deepEqual(close,{code:expectedExitCode,signal:null},`the owned native helper closed after the fixed outcome (${boundedOutcome})`);
    owner.safeCleanup=true;
    return close;
  };
  context.ownProcess(owner);
  await owner.waitForReady();
  owner.go();
  return owner;
}

function runtimeSnapshotShape(snapshot) {
  return {snapshotDigest:snapshot.snapshotDigest,releaseDigest:snapshot.releaseDigest,sizeBytes:snapshot.sizeBytes,fileCount:snapshot.fileCount,
    entries:snapshot.entries.map(entry=>({path:entry.path,digest:entry.digest,sizeBytes:entry.sizeBytes}))};
}
function assertSameRuntimeSnapshot(actual,expected,label) {
  assert.deepEqual(runtimeSnapshotShape(actual),runtimeSnapshotShape(expected),`${label}: the complete path/digest/size inventory is unchanged`);
  for(let index=0;index<expected.entries.length;index++)
    assert.ok(actual.entries[index].bytes.equals(expected.entries[index].bytes),`${label}: exact bytes remain unchanged for ${expected.entries[index].path}`);
}
async function moveOwnedCheckoutInputs(context) {
  const heldRoot=join(context.temp,'runtime-source-held'),moves=[];
  await mkdir(heldRoot);
  const inside=(base,path)=>{
    const rel=relative(resolve(base),resolve(path));
    return rel!==''&&rel!=='..'&&!rel.startsWith(`..${sep}`)&&!isAbsolute(rel);
  };
  const move=async relativePath=>{
    const source=join(context.root,...relativePath.split('/')),target=join(heldRoot,`item-${String(moves.length).padStart(3,'0')}`);
    assert.ok(inside(context.root,source)&&inside(context.temp,target),'checkout move remains inside its test-owned temporary tree');
    const info=await lstat(source);
    assert.ok(!info.isSymbolicLink(),'only physical test-owned checkout inputs are moved');
    await rename(source,target);moves.push({source,target,relativePath});
  };
  try {
    for(const path of ['apps','modules','packages','products','solutions','profiles','templates','visual','node_modules',
      'package.json','package-lock.json','tsconfig.base.json','.local/platform-builds'])await move(path);
    const requiredRuntimeTools=new Set(['build-host-preflight.ps1','build-job.ps1','runtime-supervisor.mjs']);
    for(const name of (await readdir(join(context.root,'tools'))).sort())if(!requiredRuntimeTools.has(name))await move(`tools/${name}`);
  } catch(error) {
    for(const item of [...moves].reverse())await rename(item.target,item.source);
    await rm(heldRoot,{recursive:true,force:false});
    throw error;
  }
  return {
    movedPaths:moves.map(item=>item.relativePath),
    async restore(){
      for(const item of [...moves].reverse()){
        assert.ok(!existsSync(item.source),'the source destination remains absent until its owned input is restored');
        await rename(item.target,item.source);
      }
      await rm(heldRoot,{recursive:true,force:false});
    }
  };
}

async function prepareStoppedServiceDeskRuntime(context,keyPrefix) {
  const build=await context.build(`${keyPrefix}-build`);
  const registration=await context.artifacts.register(build.id,`system:${keyPrefix}-registry`,()=>undefined);
  const digest=registration.digest;
  await context.artifacts.promote(context.product.id,context.product.version,digest,`system:${keyPrefix}-promoter`,()=>undefined);
  await context.worker.close();context.worker=undefined;
  context.runtimes=ManagedRuntimeManager.create(context.root,context.store,context.artifacts);
  const started=await context.runtimes.start({productId:context.product.id,version:context.product.version,requestKey:`${keyPrefix}-start-0001`},
    `system:${keyPrefix}-start`,()=>undefined);
  assert.equal(started.command.state,'SUCCEEDED');
  const stopped=await context.runtimes.stop({instanceId:started.instance.id,requestKey:`${keyPrefix}-stop-0001`},
    `system:${keyPrefix}-stop`,()=>undefined);
  assert.equal(stopped.command.state,'SUCCEEDED');
  assert.equal(stopped.instance.state,'STOPPED');
  assert.equal(context.store.managedRuntimeSlot().occupied,false);
  return {build,digest,started,stopped,instanceId:started.instance.id,dataDirectory:runtimeDataDirectory(context.root,started.instance.id,digest)};
}

function containsPublicValue(value,needle,{normalizePath=false}={}) {
  const target=String(needle),normalized=target.replaceAll('\\','/').toLowerCase(),seen=new Set();
  function visit(item,depth=0) {
    if(depth>8)return false;
    if(typeof item==='string') {
      const lowered=item.toLowerCase();
      if(lowered.includes(target.toLowerCase()))return true;
      if(normalizePath&&item.replaceAll('\\','/').toLowerCase().includes(normalized))return true;
      if(seen.has(item))return false;
      seen.add(item);
      const candidates=item.split(/\r?\n/).filter(line=>line.trim());
      for(const candidate of candidates) {
        if(candidate.trimStart().startsWith('{')||candidate.trimStart().startsWith('[')||candidate.trimStart().startsWith('"')) {
          try { if(visit(JSON.parse(candidate),depth+1))return true; } catch {}
        }
      }
      return false;
    }
    if(Array.isArray(item))return item.some(child=>visit(child,depth+1));
    if(item&&typeof item==='object')return Object.values(item).some(child=>visit(child,depth+1));
    return false;
  }
  return visit(value);
}

function launchRegistryOwner(root,dbPath,buildId,mode,extra='') {
  const child=spawn(process.execPath,[join(ROOT,'tests','fixtures','artifact-registry-owner.mjs'),root,dbPath,buildId,mode,String(extra)],
    {cwd:ROOT,windowsHide:true,shell:false,stdio:['pipe','pipe','pipe']});
  const owner={child,mode,stdout:'',stderr:'',closed:false,safeCleanup:false,helperClosed:false,cleanupConfirmed:false,closeResult:undefined};
  owner.closedPromise=new Promise(resolve=>child.once('close',(code,signal)=>{
    owner.closed=true;owner.safeCleanup=owner.cleanupConfirmed||owner.helperClosed;owner.closeResult={code,signal};resolve(owner.closeResult);
  }));
  child.once('error',error=>{owner.spawnError=error;owner.stderr=(owner.stderr+String(error)).slice(-32_768);});
  child.stdout.on('data',chunk=>{
    owner.stdout=(owner.stdout+chunk.toString('utf8')).slice(-32_768);
    if(owner.stdout.includes('READY_AFTER\n'))owner.helperClosed=true;
    if(owner.stdout.includes('CLEAN_CLOSED\n')||owner.stdout.includes('RELEASED\n'))owner.cleanupConfirmed=true;
  });
  child.stderr.on('data',chunk=>{owner.stderr=(owner.stderr+chunk.toString('utf8')).slice(-32_768);});
  owner.waitFor=async(marker,timeoutMs=90_000)=>{
    const end=Date.now()+timeoutMs;
    while(Date.now()<end) {
      if(owner.stdout.includes(marker))return;
      if(owner.closed)throw new Error(`Registry child closed before ${marker}: ${JSON.stringify(owner.closeResult)} ${owner.stderr} ${owner.stdout}`);
      await new Promise(resolve=>setTimeout(resolve,20));
    }
    throw new Error(`Registry child did not emit ${marker} before its deadline: ${owner.stderr} ${owner.stdout}`);
  };
  owner.waitForClose=async(timeoutMs=60_000)=>{
    let timer;
    try {
      const result=await Promise.race([owner.closedPromise,new Promise((_,reject)=>{timer=setTimeout(()=>reject(new Error(`Registry child did not close: ${owner.stderr} ${owner.stdout}`)),timeoutMs);})]);
      return result;
    } finally { clearTimeout(timer); }
  };
  owner.go=()=>{if(!owner.closed)child.stdin.write('GO\n');};
  return owner;
}

function launchStagingFileLock(executable,path) {
  const child=spawn(executable,['-NoLogo','-NoProfile','-NonInteractive','-File',join(ROOT,'tests','fixtures','hold-staging-file.ps1'),path],
    {cwd:ROOT,windowsHide:true,shell:false,stdio:['pipe','pipe','pipe']});
  const owner={child,stdout:'',stderr:'',closed:false,lockObserved:false,released:false,safeCleanup:false};
  owner.closedPromise=new Promise(resolve=>child.once('close',(code,signal)=>{
    owner.closed=true;owner.closeResult={code,signal};owner.safeCleanup=owner.lockObserved&&owner.released;resolve(owner.closeResult);
  }));
  child.once('error',error=>{owner.spawnError=error;owner.stderr=(owner.stderr+String(error)).slice(-8192);});
  child.stdout.on('data',chunk=>{
    owner.stdout=(owner.stdout+chunk.toString('utf8')).slice(-8192);
    if(/LOCKED\r?\n/.test(owner.stdout))owner.lockObserved=true;
    if(/RELEASED\r?\n/.test(owner.stdout))owner.released=true;
  });
  child.stderr.on('data',chunk=>{owner.stderr=(owner.stderr+chunk.toString('utf8')).slice(-8192);});
  child.stdin.on('error',error=>{owner.stdinError=error;owner.stderr=(owner.stderr+`stdin: ${error}`).slice(-8192);});
  owner.waitFor=async(marker,timeoutMs=15000)=>{
    const end=Date.now()+timeoutMs;
    while(Date.now()<end) {
      if(owner.stdout.includes(marker))return;
      if(owner.closed)throw new Error(`File-lock fixture closed before ${marker}: ${owner.stderr}`);
      await new Promise(resolve=>setTimeout(resolve,20));
    }
    throw new Error(`File-lock fixture did not emit ${marker}: ${owner.stderr}`);
  };
  owner.releaseRequested=false;
  owner.release=()=>{
    if(owner.releaseRequested||owner.closed)return;
    owner.releaseRequested=true;
    try { child.stdin.write('GO\n'); }
    catch(error) { owner.stdinError=error;owner.stderr=(owner.stderr+`stdin: ${error}`).slice(-8192); }
  };
  owner.waitForClose=async(timeoutMs=25000)=>{
    let timer;
    try { return await Promise.race([owner.closedPromise,new Promise((_,reject)=>{timer=setTimeout(()=>reject(new Error(`File-lock fixture did not close: ${owner.stderr}`)),timeoutMs);})]); }
    finally { clearTimeout(timer); }
  };
  return owner;
}

    await registrationLock.waitFor('LOCKED');
    const originalBegin=context.store.beginArtifactPublication.bind(context.store);
    let registrationEntryAt,registrationFinalAuthorizationAt;
    context.store.beginArtifactPublication=(intent,authorize,observe)=>{
      registrationEntryAt=Date.now();
      return originalBegin(intent,()=>{registrationFinalAuthorizationAt=Date.now();return authorize?.();},observe);
    };
    try {
      const registrationCookie=await pair(context.operator);
      const sessionResponse=await pairedFetch(context.operator,'/platform-api/session',{cookie:registrationCookie});
      const registrationExpiresAt=(await sessionResponse.json()).expiresAt;
      const operationCount=context.store.artifactOperationCount(),auditCount=context.store.auditLog(500).length;
      const staging=(await readdir(context.artifacts.stagingRoot)).sort();
      const started=Date.now();
      const response=await pairedFetch(context.operator,'/platform-api/artifacts/register',{method:'POST',cookie:registrationCookie,
        body:{confirm:true,buildId:secondBuild.id}});
      const elapsed=Date.now()-started;
      assert.equal(response.status,401,await response.text());
      await registrationLock.waitFor('RELEASED');
      assert.ok(registrationEntryAt<registrationExpiresAt,'the original session was alive when build input verification reached the registration transaction');
      assert.ok(registrationFinalAuthorizationAt>=registrationExpiresAt,'registration authorization ran again after BEGIN IMMEDIATE released');
      assert.ok(elapsed>=2100,`registration waited for the independent SQLite writer (${elapsed}ms)`);
      assert.equal(context.store.artifactOperationCount(),operationCount,'expired final registration authorization writes no operation');
      assert.equal(context.store.auditLog(500).length,auditCount,'expired final registration authorization writes no audit');
      assert.equal(context.store.artifactRegistrySlot().operationId,null,'expired registration reserves no physical slot');
      assert.deepEqual((await readdir(context.artifacts.stagingRoot)).sort(),staging,'expired registration creates no staging tree');
      assert.equal(context.store.artifactRecord(secondBuild.snapshotDigest),undefined);
    } finally { context.store.beginArtifactPublication=originalBegin; }
    assert.equal((await registrationLock.waitForClose()).code,0,registrationLock.stderr);
    assert.equal(registrationLock.safeCleanup,true,'the external SQLite lock owner closed before fixture cleanup');

    const identity=context.store.artifactBuildIdentity(secondBuild.id);
    const interrupted=context.ownProcess(launchRegistryOwner(context.root,context.dbPath,secondBuild.id,'pause-after'));owners.push(interrupted);
    await interrupted.waitFor('READY_AFTER');
    const objectPath=join(context.artifacts.objectsRoot,identity.snapshotDigest);
    const moved=readBoundedSnapshot(objectPath,identity,ARTIFACT_REGISTRY_LIMITS);
    assert.equal(moved.snapshotDigest,identity.snapshotDigest,'the helper closed after moving a complete object before the owner was interrupted');
    const interruptedSlot=context.store.artifactRegistrySlot();
    assert.ok(interruptedSlot.operationId);
    assert.equal(context.store.artifactRecord(identity.snapshotDigest),undefined);
    assert.equal(interrupted.child.kill(),true,'the paused owner is killed only after its publisher helper has closed');
    const interruptedClose=await interrupted.waitForClose();
    assert.notEqual(interruptedClose.code,0,'the owner process was interrupted before it could persist its move observation');
    assert.equal(interrupted.helperClosed,true,'READY_AFTER follows the publisher nonce proof and helper close');
    assert.equal(interrupted.safeCleanup,true,'the owned Node process closed after the publisher helper had already closed');
    assert.deepEqual({...context.store.artifactRegistrySlot()},{...interruptedSlot},'a process crash never releases its full durable reservation');
    assert.equal(context.store.artifactRecord(identity.snapshotDigest),undefined,'the moved object is not advertised by the interrupted owner');
    const interruptedOperation=(await context.artifacts.list()).operations.find(row=>row.id===interruptedSlot.operationId);
    assert.ok(interruptedOperation);
    assert.equal(interruptedOperation.ownerObservation,'not-observed-in-this-process');
    assert.equal(interruptedOperation.destinationVerified,false,'physical movement alone is not durable destination verification');
    const adopted=await context.artifacts.register(secondBuild.id,'system:artifact-process-adopter',()=>undefined);
    assert.equal(adopted.digest,identity.snapshotDigest,'a fresh explicit verification can adopt the complete orphan');
    assert.deepEqual({...context.store.artifactRegistrySlot()},{...interruptedSlot},'metadata-only orphan adoption keeps the older unknown owner reservation intact');

    const lock=context.ownProcess(launchRegistryOwner(context.root,context.dbPath,'','hold-lock',2600));owners.push(lock);
    await lock.waitFor('LOCKED');
    const originalPromote=context.store.promoteArtifact.bind(context.store);
    let mutationEntryAt,finalAuthorizationAt;
    context.store.promoteArtifact=(input,authorize)=>{
      mutationEntryAt=Date.now();
      return originalPromote(input,()=>{finalAuthorizationAt=Date.now();return authorize?.();});
    };
    const cookie=await pair(context.operator);
    const sessionResponse=await pairedFetch(context.operator,'/platform-api/session',{cookie});
    const expiresAt=(await sessionResponse.json()).expiresAt;
    const auditBeforeExpiry=context.store.auditLog(500).length;
    const started=Date.now();
    const expiredMutation=await pairedFetch(context.operator,'/platform-api/releases',{method:'POST',cookie,
      body:{confirm:true,productId:identity.productId,version:identity.version,artifactDigest:identity.snapshotDigest}});
    const elapsed=Date.now()-started;
    const mutationBody=await expiredMutation.text();
    assert.equal(expiredMutation.status,401,`status=${expiredMutation.status} elapsed=${elapsed} mutationEntryAt=${mutationEntryAt} finalAuthorizationAt=${finalAuthorizationAt} expiresAt=${expiresAt} lock=${lock.stdout} body=${mutationBody}`);
    await lock.waitFor('RELEASED');
    assert.ok(mutationEntryAt<expiresAt,`the original session was live at the store mutation boundary (${mutationEntryAt} < ${expiresAt})`);
    assert.ok(finalAuthorizationAt>=expiresAt,`the final in-transaction authorization ran after expiry (${finalAuthorizationAt} >= ${expiresAt})`);
    assert.ok(elapsed>=2100,`request waited for the independent BEGIN IMMEDIATE holder (${elapsed}ms)`);
    context.store.promoteArtifact=originalPromote;
    const lockClose=await lock.waitForClose();
    assert.equal(lockClose.code,0,lock.stderr);
    assert.equal(lock.safeCleanup,true,'the external SQLite lock owner closed before fixture cleanup');
    assert.equal(context.store.artifactReleases().some(row=>row.productId===identity.productId&&row.version===identity.version),false,
      'the original session expired while SQLite BEGIN was blocked, so no release was promoted');
    assert.equal(context.store.auditLog(500).length,auditBeforeExpiry,'expired final authorization writes no release audit');
  } finally {
    for(const owner of owners)if(!owner.closed&&['pause-before','pause-after'].includes(owner.mode))owner.go();
    for(const owner of owners)if(!owner.closed)await owner.waitForClose();
  }
});

function runPublisher(executable,script,registryRoot,operationId,digest,nonce) {
  const args=['-NoLogo','-NoProfile','-NonInteractive','-File',script,registryRoot,operationId,digest,nonce];
  return new Promise((resolve,reject)=>execFile(executable,args,
    {cwd:ROOT,windowsHide:true,shell:false,timeout:15000,maxBuffer:8192,encoding:'utf8'},
    (error,stdout,stderr)=>{
      if(error&&typeof error.code!=='number') { reject(error); return; }
      resolve({code:error?error.code:0,signal:error?.signal??null,stdout,stderr,error:error?.message??null});
    }));
}

test('fixed registry publisher moves into vacant paths and preserves existing empty directories, populated directories, and files', {skip:!WINDOWS}, async t=>{
  const temp=await mkdtemp(join(tmpdir(),'qb-publisher-no-overwrite-'));
  t.after(async()=>{await rm(temp,{recursive:true,force:true,maxRetries:4,retryDelay:100});});
  const host=JSON.parse(await readFile(join(ROOT,'.local','platform-build-host.json'),'utf8'));
  const script=join(ROOT,'tools','registry-publish.ps1');
  const cases=[
    {name:'empty-directory',kind:'directory'},
    {name:'nonempty-directory',kind:'populated'},
    {name:'regular-file',kind:'file'}
  ];
  for(const [index,item] of cases.entries()) {
    const registryRoot=join(temp,item.name),operationId=randomUUID(),digest=String(index+1).repeat(64),nonce=String(index+1).repeat(24);
    const source=join(registryRoot,'staging',operationId,'payload'),destination=join(registryRoot,'objects',digest);
    await mkdir(source,{recursive:true});await mkdir(join(registryRoot,'objects'),{recursive:true});
    await writeFile(join(source,'payload.txt'),'new-payload');
    if(item.kind==='directory')await mkdir(destination);
    if(item.kind==='populated'){await mkdir(destination);await writeFile(join(destination,'sentinel.txt'),'preserve-me');}
    if(item.kind==='file')await writeFile(destination,'preserve-me');
    const result=await runPublisher(host.path,script,registryRoot,operationId,digest,nonce);
    assert.equal(result.code,3,`${item.name}: ${result.stderr}`);
    assert.match(result.stdout,new RegExp(`QB_REGISTRY_MOVE:${nonce}:EXISTS`));
    assert.equal(await readFile(join(source,'payload.txt'),'utf8'),'new-payload','occupied destinations never consume the staged source');
    if(item.kind==='populated')assert.equal(await readFile(join(destination,'sentinel.txt'),'utf8'),'preserve-me');
    if(item.kind==='file')assert.equal(await readFile(destination,'utf8'),'preserve-me');
  }
  const registryRoot=join(temp,'vacant'),operationId=randomUUID(),digest=createHash('sha256').update('vacant').digest('hex'),nonce='Z'.repeat(24);
  const source=join(registryRoot,'staging',operationId,'payload'),destination=join(registryRoot,'objects',digest);
  await mkdir(source,{recursive:true});await mkdir(join(registryRoot,'objects'),{recursive:true});
  await writeFile(join(source,'verified.txt'),'complete bytes');
  const result=await runPublisher(host.path,script,registryRoot,operationId,digest,nonce);
  assert.equal(result.code,0, result.stderr);
  assert.match(result.stdout,new RegExp(`QB_REGISTRY_MOVE:${nonce}:MOVED`));
  assert.equal(await readFile(join(destination,'verified.txt'),'utf8'),'complete bytes');
});

test('registry publisher proof parsing rejects ambiguous, mismatched, and oversized output',()=>{
  const nonce='A'.repeat(24),marker=`QB_REGISTRY_MOVE:${nonce}:MOVED`;
  assert.equal(parseRegistryPublisherTranscript(`${marker}\n`,nonce,0,null),'MOVED');
  assert.equal(parseRegistryPublisherTranscript(`QB_REGISTRY_MOVE:${nonce}:EXISTS\n`,nonce,3,null),'EXISTS');
  assert.equal(parseRegistryPublisherTranscript(`QB_REGISTRY_MOVE:${nonce}:FAILED\n`,nonce,1,null),'FAILED');
  assert.equal(parseRegistryPublisherTranscript(`noise\n${marker}\n`,nonce,0,null),'UNKNOWN');
  assert.equal(parseRegistryPublisherTranscript(`${marker}\n${marker}\n`,nonce,0,null),'UNKNOWN');
  assert.equal(parseRegistryPublisherTranscript(`noise\n${marker}\n`,nonce,0,null,true),'UNKNOWN',
    'a valid suffix cannot erase evidence that the bounded stdout transcript overflowed');
  assert.equal(parseRegistryPublisherTranscript(`${marker}\n`,nonce,3,null),'UNKNOWN','proof text and process exit code must agree');
});
